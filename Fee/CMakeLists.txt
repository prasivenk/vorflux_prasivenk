cmake_minimum_required(VERSION 3.10)
project(AUTOSAR_Fee C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Werror -pedantic")
include_directories(include)
add_library(fee STATIC src/Fee.c src/Fee_Internal.c src/Fee_Sector.c src/Fee_Cfg.c src/Det.c src/MemAcc.c)
add_library(unity STATIC test/unity/unity.c)
target_include_directories(unity PUBLIC test/unity)
add_executable(test_fee test/test_Fee.c)
target_link_libraries(test_fee fee unity)
add_executable(test_fee_internal test/test_Fee_Internal.c)
target_link_libraries(test_fee_internal fee unity)
add_executable(test_fee_sector test/test_Fee_Sector.c)
target_link_libraries(test_fee_sector fee unity)
add_executable(test_memacc test/test_MemAcc.c)
target_link_libraries(test_memacc fee unity)
enable_testing()
add_test(NAME TestFee COMMAND test_fee)
add_test(NAME TestFeeInternal COMMAND test_fee_internal)
add_test(NAME TestFeeSector COMMAND test_fee_sector)
add_test(NAME TestMemAcc COMMAND test_memacc)
